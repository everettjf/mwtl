Quick start

Hello, native window.

The shortest example uses a CRTP window class, native controls, and a one-line application entry point. Common messages are ordinary C++20 member functions—no message-map macros are required.

#include <mwtl/mwtl.h>

class MainWindow final : public mwtl::Window<MainWindow> {
public:
    void BuildUI() {
        SetTitle(L"mwtl hello");
        label_.Create(*this, mwtl::ControlId{100}, L"Hello from modern WTL", {24, 24, 260, 28});
        close_.Create(*this, mwtl::ControlId{101}, L"Close", {24, 64, 100, 32});
    }

    void OnCommand(const mwtl::CommandEvent& event) {
        if (event.IsClicked(mwtl::ControlId{101})) {
            Close();
        }
    }

private:
    mwtl::Label label_;
    mwtl::Button close_;
};

int WINAPI wWinMain(HINSTANCE instance, HINSTANCE, PWSTR, int show) {
    return mwtl::RunApplication<MainWindow>(instance, show);
}

Target: mwtl_hello. The final executable embeds the Unicode/Common Controls v6 and Per-Monitor V2 manifest declarations.
