Quick start

Hello, native window.

The shortest example needs only a CRTP window class and an Application entry point.

#include <mwtl/mwtl.h>

class MainWindow final : public mwtl::Window<MainWindow> {
public:
    void BuildUI() { SetTitle(L"mwtl hello"); }
};

int WINAPI wWinMain(HINSTANCE instance, HINSTANCE, PWSTR, int show) {
    try {
        return mwtl::Application(instance).Run<MainWindow>(show);
    } catch (...) {
        return EXIT_FAILURE;
    }
}

Target: mwtl_hello. The final executable embeds the Unicode/Common Controls v6 and Per-Monitor V2 manifest declarations.
