Quick start

Hello, native window.

The shortest example names its window class once, uses native controls, and has a one-line application entry point. Common messages are ordinary C++20 overrides—no CRTP repetition or message-map macros are required.

#include <mwtl/mwtl.h>

class MainWindow final : public mwtl::WindowBase {
public:
    void BuildUI() override {
        SetTitle(L"mwtl hello");
        label_.Create(*this, mwtl::ControlId{100}, L"Hello from modern WTL", {24, 24, 260, 28});
        close_.Create(*this, mwtl::ControlId{101}, L"Close", {24, 64, 100, 32});
    }

    mwtl::EventResult OnCommand(const mwtl::CommandEvent& event) override {
        if (event.IsClicked(close_)) {
            static_cast<void>(Close());
            return mwtl::EventResult::Handled();
        }
        return mwtl::EventResult::Propagate();
    }

private:
    mwtl::Label label_;
    mwtl::Button close_;
};

int WINAPI wWinMain(HINSTANCE instance, HINSTANCE, PWSTR, int show) {
    return mwtl::RunApplication<MainWindow>(instance, show);
}

Target: mwtl_hello. The final executable embeds the Unicode/Common Controls v6 and Per-Monitor V2 manifest declarations.
